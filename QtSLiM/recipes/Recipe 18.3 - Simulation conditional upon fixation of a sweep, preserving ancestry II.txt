// Keywords: tree-sequence recording, tree sequence recording, conditional sweep

initialize() {
	initializeTreeSeq();
	initializeMutationRate(1e-8);
	initializeMutationType("m2", 0.5, "g", -0.01, 1.0);  // deleterious
	initializeMutationType("m3", 1.0, "f", 0.05);        // introduced
	initializeGenomicElementType("g1", m2, 1.0);
	initializeGenomicElement(g1, 0, 99999);
	initializeRecombinationRate(1e-8);
}
1 early() {
	defineConstant("simID", getSeed());
	sim.addSubpop("p1", 500);
}
1000 late() {
	// generate the side values to be preserved
	inds = sim.subpopulations.individuals;
	inds.setValuesVectorized("X", rdunif(size(inds), 0, 100000));
	
	// record the side values and pedigree IDs in metadata
	inds = sortBy(sim.subpopulations.individuals, "pedigreeID");
	metadataDict = Dictionary("X", inds.getValue("X"), "ids", inds.pedigreeID);
	
	// print the side values, so we can check them visually
	catn("\nSAVED VALUES:");
	print(inds.getValue("X"));
	
	target = sample(p1.haplosomes, 1);
	target.addNewDrawnMutation(m3, 10000);
	defineConstant("PATH", tempdir() + "slim_" + simID + ".trees");
	sim.treeSeqOutput(PATH, metadata=metadataDict);
}
1000:100000 late() {
	if (sim.countOfMutationsOfType(m3) == 0) {
		if (sum(sim.substitutions.mutationType == m3) == 1) {
			cat(simID + ": FIXED\n");
			sim.treeSeqOutput("slim_" + simID + "_FIXED.trees");
			sim.simulationFinished();
		} else {
			cat(simID + ": LOST - RESTARTING\n");
			sim.readFromPopulationFile(PATH);
			
			// restore side values from metadata back into the individuals
			inds = sortBy(sim.subpopulations.individuals, "pedigreeID");
			inds.setValuesVectorized("X", treeSeqMetadata(PATH).getValue("X"));
			
			// print the restored side values, so we can check them visually
			catn("\nRESTORED VALUES:");
			print(inds.getValue("X"));
		}
	}
}
